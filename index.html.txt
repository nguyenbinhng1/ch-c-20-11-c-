      function initAudio() {
        try {
          if (typeof Tone !== 'undefined') {
            if (!masterFilter) {
              masterFilter = new Tone.Filter({
                frequency: 3600,
                type: 'lowpass',
                rolloff: -12
              }).toDestination();
            }
            if (!chimeSynth) {
              chimeSynth = new Tone.PolySynth(Tone.Synth, {
                oscillator: { type: 'triangle' },
                envelope: { attack: 0.025, decay: 0.25, sustain: 0.2, release: 1.1 },
                volume: -8
              }).connect(masterFilter);
            }
            if (!pianoSynth) {
              pianoSynth = new Tone.PolySynth(Tone.Synth, {
                oscillator: { type: 'triangle' },
                envelope: { attack: 0.015, decay: 0.65, sustain: 0.16, release: 1.3 },
                volume: -4
              }).connect(masterFilter);
            }
            if (!padSynth) {
              padSynth = new Tone.PolySynth(Tone.Synth, {
                oscillator: { type: 'sine' },
                envelope: { attack: 0.45, decay: 0.4, sustain: 0.55, release: 1.8 },
                volume: -16
              }).connect(masterFilter);
            }
            return 'tone';
          }

          // Dự phòng bằng Web Audio API gốc nếu CDN Tone.js chưa tải được
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          if (!AudioCtx) return null;
          if (!nativeAudioCtx) {
            nativeAudioCtx = new AudioCtx();
          }
          return 'native';
        } catch (err) {
          console.warn('Không thể khởi tạo âm thanh:', err);
          return null;
        }
      }

      function playNativeNote(noteStr, durationSec, gainLevel, oscType = 'triangle', delaySec = 0) {
        if (!nativeAudioCtx) return;
        try {
          const now = nativeAudioCtx.currentTime + delaySec;
          const osc = nativeAudioCtx.createOscillator();
          const gain = nativeAudioCtx.createGain();

          osc.type = oscType;
          osc.frequency.setValueAtTime(noteToFreq(noteStr), now);

          gain.gain.setValueAtTime(0.0001, now);
          gain.gain.exponentialRampToValueAtTime(Math.max(gainLevel, 0.001), now + 0.02);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + durationSec);

          osc.connect(gain);
          gain.connect(nativeAudioCtx.destination);
          osc.start(now);
          osc.stop(now + durationSec + 0.05);
        } catch (err) {
          console.warn('Lỗi phát nốt dự phòng:', err);
        }
      }

            if (stepInBar === 0) {
              padSynth.triggerAttackRelease(bar.pad, '1m', now, 0.55);
            }
            const arpNote = bar.arp[stepInBar];
            if (arpNote) {
              pianoSynth.triggerAttackRelease(arpNote, '4n', now, 0.32);
            }
            const mel = bar.melody[stepInBar];

              pianoSynth.triggerAttackRelease(mel.note, mel.dur, now, 0.85);

          } else {

            }
            const arpNote = bar.arp[stepInBar];
            if (arpNote) {

            }
            const mel = bar.melody[stepInBar];
            if (mel) {
              playNativeNote(mel.note, mel.sec || 0.55, 0.16, 'triangle');
            }
          }

          musicStep = (musicStep + 1) % totalSteps;
        } catch (err) {
          console.warn('Lỗi nhịp phát nhạc:', err);
        }
      }

          if (engine === 'tone') {
            await Tone.start();
            if (Tone.context && Tone.context.state !== 'running') {
              await Tone.context.resume();

          } else if (engine === 'native' && nativeAudioCtx && nativeAudioCtx.state === 'suspended') {
            await nativeAudioCtx.resume();
          }

          if (!musicPlaying || !engine) return;

          if (musicTimer) clearInterval(musicTimer);
          playMusicStep();
          musicTimer = setInterval(playMusicStep, 375);

          if (!isAuto) {

          }
        } catch (err) {
          musicPlaying = false;
          updateMusicUI();
          console.warn('Không thể bật nhạc nền:', err);
        }
      }

      function playChime(chordType = 'celebrate') {
        if (!soundEnabled) return;
        try {
          const engine = initAudio();
          if (!engine) return;

          if (engine === 'tone') {
            const now = Tone.now();
            if (chordType === 'celebrate') {
              const notes = ['C5', 'E5', 'G5', 'B5', 'C6'];
              notes.forEach((note, idx) => {
                chimeSynth.triggerAttackRelease(note, '8n', now + idx * 0.09, 0.55);
              });
            } else if (chordType === 'spark') {
              chimeSynth.triggerAttackRelease(['G5', 'B5', 'D6'], '16n', now, 0.45);
            } else if (chordType === 'heart') {
              chimeSynth.triggerAttackRelease('A5', '16n', now, 0.45);
              chimeSynth.triggerAttackRelease('E6', '16n', now + 0.08, 0.45);
            }
          } else {
            if (chordType === 'celebrate') {
              ['C5', 'E5', 'G5', 'B5', 'C6'].forEach((n, idx) => {
                playNativeNote(n, 0.45, 0.1, 'triangle', idx * 0.09);
              });
            } else if (chordType === 'spark') {
              ['G5', 'B5', 'D6'].forEach((n) => playNativeNote(n, 0.25, 0.08, 'triangle'));
            } else if (chordType === 'heart') {
              playNativeNote('A5', 0.25, 0.09, 'triangle', 0);
              playNativeNote('E6', 0.28, 0.09, 'triangle', 0.08);
            }
          }
        } catch (err) {
          console.warn('Lỗi phát âm thanh:', err);
        }
      }

      function renderBgFrame() {
        if (!bgCtx) return;
        bgCtx.clearRect(0, 0, bgWidth, bgHeight);

        // Vẽ các công thức Vật Lí trôi nhẹ nhàng
        bgCtx.textAlign = 'center';
        for (let i = 0; i < bgFormulas.length; i++) {
          const f = bgFormulas[i];
          f.y += f.vy;
          if (f.y < -30) {
            f.y = bgHeight + 30;
            f.x = Math.random() * bgWidth;
          }
          bgCtx.font = `italic ${f.size}px "Playfair Display", serif`;
          bgCtx.fillStyle = `rgba(253, 230, 138, ${f.opacity})`;
          bgCtx.fillText(f.text, f.x, f.y);
        }

        // Vẽ các hạt photon & đường liên kết trường lực
        for (let i = 0; i < bgParticles.length; i++) {
          const p = bgParticles[i];
          p.x += p.vx;
          p.y += p.vy;

          if (p.x < 0 || p.x > bgWidth) p.vx *= -1;
          if (p.y < 0 || p.y > bgHeight) p.vy *= -1;

          // Tương tác nhẹ với vị trí con trỏ
          const dxMouse = pointer.x - p.x;
          const dyMouse = pointer.y - p.y;
          const distMouse = Math.hypot(dxMouse, dyMouse);
          if (distMouse < 130 && distMouse > 1) {
            p.x -= (dxMouse / distMouse) * 0.25;
            p.y -= (dyMouse / distMouse) * 0.25;
          }

          bgCtx.beginPath();
          bgCtx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
          bgCtx.fillStyle = `hsla(${p.hue}, 92%, 72%, 0.7)`;
          bgCtx.fill();

          // Nối dây các hạt gần nhau
          for (let j = i + 1; j < bgParticles.length; j++) {
            const p2 = bgParticles[j];
            const dist = Math.hypot(p.x - p2.x, p.y - p2.y);
            if (dist < 115) {
              const alpha = (1 - dist / 115) * 0.16;
              bgCtx.beginPath();
              bgCtx.moveTo(p.x, p.y);
              bgCtx.lineTo(p2.x, p2.y);
              bgCtx.strokeStyle = `rgba(148, 163, 184, ${alpha})`;
              bgCtx.lineWidth = 0.8;
              bgCtx.stroke();
            }
          }
        }

      // -------------------- 4. CANVAS PHÁO HOA QUANG PHỔ --------------------
      const fwCanvas = document.getElementById('fireworks-canvas');
      const fwCtx = fwCanvas ? fwCanvas.getContext('2d') : null;
      let fwWidth = 0;
      let fwHeight = 0;
      let fwSparks = [];

      const SPECTRUM_COLORS = [
        '#fbbf24', '#f97316', '#ef4444', '#22c55e', '#38bdf8', '#818cf8', '#e879f9', '#fef08a'
      ];

      function resizeFwCanvas() {
        if (!fwCanvas || !fwCtx) return;
        const dpr = Math.min(window.devicePixelRatio || 1, 2);
        fwWidth = window.innerWidth;
        fwHeight = window.innerHeight;
        fwCanvas.width = fwWidth * dpr;
        fwCanvas.height = fwHeight * dpr;
        fwCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
      }

      function launchFireworkBurst(x, y, count = 42) {
        const cx = typeof x === 'number' ? x : fwWidth * (0.2 + Math.random() * 0.6);
        const cy = typeof y === 'number' ? y : fwHeight * (0.18 + Math.random() * 0.35);
        for (let i = 0; i < count; i++) {
          const angle = (Math.PI * 2 * i) / count + (Math.random() - 0.5) * 0.25;
          const speed = Math.random() * 4.5 + 1.8;
          fwSparks.push({
            x: cx,
            y: cy,
            vx: Math.cos(angle) * speed,
            vy: Math.sin(angle) * speed,
            life: 1,
            decay: Math.random() * 0.015 + 0.012,
            size: Math.random() * 2.6 + 1.4,
            color: SPECTRUM_COLORS[Math.floor(Math.random() * SPECTRUM_COLORS.length)]
          });
        }
      }

      function triggerCelebrationSalvo() {
        playChime('celebrate');
        launchFireworkBurst(fwWidth * 0.28, fwHeight * 0.28, 45);
        setTimeout(() => launchFireworkBurst(fwWidth * 0.72, fwHeight * 0.24, 45), 220);
        setTimeout(() => launchFireworkBurst(fwWidth * 0.5, fwHeight * 0.34, 58), 460);
      }

      function renderFireworksFrame() {
        if (!fwCtx) return;
        fwCtx.clearRect(0, 0, fwWidth, fwHeight);

        for (let i = fwSparks.length - 1; i >= 0; i--) {
          const s = fwSparks[i];
          s.x += s.vx;
          s.y += s.vy;
          s.vy += 0.055; // Gia tốc trọng trường g
          s.vx *= 0.985;
          s.life -= s.decay;

          if (s.life <= 0) {
            fwSparks.splice(i, 1);
            continue;
          }

          fwCtx.save();
          fwCtx.globalAlpha = Math.max(s.life, 0);
          fwCtx.beginPath();
          fwCtx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
          fwCtx.fillStyle = s.color;
          fwCtx.shadowColor = s.color;
          fwCtx.shadowBlur = 10;
          fwCtx.fill();
          fwCtx.restore();
        }
      }

      // -------------------- 5. PHÒNG THÍ NGHIỆM LĂNG KÍNH TÁN SẮC 20/11 --------------------
      const prismCanvas = document.getElementById('prism-canvas');
      const prismContainer = document.getElementById('prism-container');
      const prismCtx = prismCanvas ? prismCanvas.getContext('2d') : null;

      let prismWidth = 600;
      let prismHeight = 360;
      let incidentAngleDeg = 34;
      let baseRefractiveIndex = 1.52;
      let autoSweep = false;
      let sweepPhase = 0;
      let prismPulse = 0;

      // 7 màu quang phổ gắn với 7 lời chúc tặng Cô Nguyễn Thị Thuý
      const RAINBOW_WISHES = [
        { name: 'Nhiệt Huyết', color: '#ff4d4d', deltaN: -0.045 },
        { name: 'Hạnh Phúc', color: '#ff944d', deltaN: -0.028 },
        { name: 'Thành Công', color: '#ffd633', deltaN: -0.012 },
        { name: 'Sức Khỏe', color: '#4ade80', deltaN: 0.004 },
        { name: 'Bình An', color: '#38bdf8', deltaN: 0.022 },
        { name: 'Vững Bước', color: '#818cf8', deltaN: 0.042 },
        { name: 'Yêu Thương', color: '#d946ef', deltaN: 0.064 }
      ];

      function resizePrismCanvas() {
        if (!prismCanvas || !prismContainer || !prismCtx) return;
        const rect = prismContainer.getBoundingClientRect();
        const dpr = Math.min(window.devicePixelRatio || 1, 2);
        prismWidth = Math.max(rect.width, 280);
        prismHeight = Math.max(rect.height, 260);
        prismCanvas.width = prismWidth * dpr;
        prismCanvas.height = prismHeight * dpr;
        prismCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
      }

      function updatePrismReadout() {
        const angleReadout = document.getElementById('angle-readout');
        const indexReadout = document.getElementById('index-readout');
        const incidentValText = document.getElementById('incident-val-text');
        const dispersionValText = document.getElementById('dispersion-val-text');
        const incidentSlider = document.getElementById('incident-slider');

        const roundedAngle = Math.round(incidentAngleDeg);
        if (angleReadout) angleReadout.textContent = `${roundedAngle}°`;
        if (incidentValText) incidentValText.textContent = `${roundedAngle}°`;
        if (indexReadout) indexReadout.textContent = baseRefractiveIndex.toFixed(2);
        if (dispersionValText) dispersionValText.textContent = baseRefractiveIndex.toFixed(2);
        if (incidentSlider && autoSweep) incidentSlider.value = roundedAngle;
      }

      function renderPrismFrame() {
        if (!prismCtx) return;

        if (autoSweep) {
          sweepPhase += 0.022;
          incidentAngleDeg = 38 + Math.sin(sweepPhase) * 20;
          updatePrismReadout();
        }
        if (prismPulse > 0) prismPulse *= 0.94;

        const w = prismWidth;
        const h = prismHeight;
        prismCtx.clearRect(0, 0, w, h);

        // Nền lưới tọa độ phòng thí nghiệm tinh tế
        prismCtx.strokeStyle = 'rgba(148, 163, 184, 0.06)';
        prismCtx.lineWidth = 1;
        const gridStep = 28;
        for (let gx = 0; gx < w; gx += gridStep) {
          prismCtx.beginPath();
          prismCtx.moveTo(gx, 0);
          prismCtx.lineTo(gx, h);
          prismCtx.stroke();
        }
        for (let gy = 0; gy < h; gy += gridStep) {
          prismCtx.beginPath();
          prismCtx.moveTo(0, gy);
          prismCtx.lineTo(w, gy);
          prismCtx.stroke();
        }

        // Đặt lăng kính lớn và dịch sang trái một chút để chùm 7 tia cầu vồng tỏa rộng, dễ nhìn trên điện thoại
        const isSmall = w < 540;
        const cx = isSmall ? w * 0.23 : w * 0.31;
        const cy = h * 0.5;
        const side = Math.min(w * (isSmall ? 0.3 : 0.24), h * 0.35);
        const triH = (Math.sqrt(3) / 2) * side;

        const topV = { x: cx, y: cy - triH * 0.58 };
        const leftV = { x: cx - side / 2, y: cy + triH * 0.42 };
        const rightV = { x: cx + side / 2, y: cy + triH * 0.42 };

        // 1. Tính điểm tia sáng trắng chạm mặt bên trái của lăng kính và vẽ tia tới
        const hitT = 0.46;
        const hitX = topV.x + (leftV.x - topV.x) * hitT;
        const hitY = topV.y + (leftV.y - topV.y) * hitT;

        const incRad = (incidentAngleDeg * Math.PI) / 180;
        const edgeDx = leftV.x - topV.x;
        const edgeDy = leftV.y - topV.y;
        const normalAngle = Math.atan2(edgeDx, -edgeDy);
        const sourceAngle = normalAngle - incRad;

        const incBeamLen = Math.hypot(w, h);
        const startX = hitX + Math.cos(sourceAngle) * incBeamLen;
        const startY = hitY + Math.sin(sourceAngle) * incBeamLen;

        // Vẽ đường pháp tuyến nét đứt tại điểm tới
        prismCtx.save();
        prismCtx.setLineDash([5, 5]);
        prismCtx.strokeStyle = 'rgba(251, 191, 36, 0.35)';
        prismCtx.lineWidth = 1.2;
        prismCtx.beginPath();
        prismCtx.moveTo(hitX + Math.cos(normalAngle) * 44, hitY + Math.sin(normalAngle) * 44);
        prismCtx.lineTo(hitX - Math.cos(normalAngle) * 44, hitY - Math.sin(normalAngle) * 44);
        prismCtx.stroke();
        prismCtx.restore();

        // Vẽ chùm tia sáng trắng chiếu vào lăng kính
        prismCtx.save();
        prismCtx.strokeStyle = 'rgba(255, 255, 255, 0.25)';
        prismCtx.lineWidth = 10 + prismPulse * 4;
        prismCtx.lineCap = 'round';
        prismCtx.beginPath();
        prismCtx.moveTo(startX, startY);
        prismCtx.lineTo(hitX, hitY);
        prismCtx.stroke();

        prismCtx.strokeStyle = '#ffffff';
        prismCtx.lineWidth = 3.5 + prismPulse * 1.5;
        prismCtx.shadowColor = '#fef08a';
        prismCtx.shadowBlur = 14;
        prismCtx.beginPath();
        prismCtx.moveTo(startX, startY);
        prismCtx.lineTo(hitX, hitY);
        prismCtx.stroke();
        prismCtx.restore();

        // 2. Tính toán 7 chùm tia tán sắc chuyển động tự do, mượt mà theo góc tới (i) và chiết suất (n)
        const effPrismAngle = 0.74;
        const sinR1Base = Math.sin(incRad) / baseRefractiveIndex;
        const r1Base = Math.asin(Math.min(Math.max(sinR1Base, -0.95), 0.95));
        const r2Base = effPrismAngle - r1Base;
        const sinI2Base = Math.min(Math.max(baseRefractiveIndex * Math.sin(r2Base), -0.95), 0.95);
        const i2Base = Math.asin(sinI2Base);

        // Góc nghiêng toàn chùm cầu vồng thay đổi tự nhiên khi kéo góc tới (i)
        const rawTilt = (i2Base - 0.52) * (isSmall ? 0.34 : 0.26);
        const baseTilt = Math.min(Math.max(rawTilt, -0.16), 0.16);

        // Độ mở rộng dải quang phổ thay đổi rõ rệt theo chiết suất (n) và luôn tách đều không chạm chữ
        const nNorm = Math.min(Math.max((baseRefractiveIndex - 1.35) / 0.5, 0), 1);
        const angleStep = isSmall ? 0.125 + nNorm * 0.04 : 0.08 + nNorm * 0.04;

        const padX = isSmall ? 10 : 20;
        const padY = 24;
        const rayDrawLen = Math.hypot(w, h) + 80;

        const raysData = RAINBOW_WISHES.map((ray, idx) => {
          const offset = idx - 3;
          const exitT = Math.min(Math.max(0.47 + offset * 0.052 + (r1Base - 0.35) * 0.2, 0.15), 0.87);
          const exitX = topV.x + (rightV.x - topV.x) * exitT;
          const exitY = topV.y + (rightV.y - topV.y) * exitT;

          const rayAngle = baseTilt + offset * angleStep;
          const cosA = Math.cos(rayAngle);
          const sinA = Math.sin(rayAngle);
          const endX = exitX + cosA * rayDrawLen;
          const endY = exitY + sinA * rayDrawLen;

          // Tính đúng khoảng cách tối đa dọc theo tia trước khi chạm mép khung để chữ luôn gọn trong màn hình
          const tMaxX = (w - padX - exitX) / cosA;
          let tMaxY = Infinity;
          if (sinA > 0.001) {
            tMaxY = (h - padY - exitY) / sinA;
          } else if (sinA < -0.001) {
            tMaxY = (padY - exitY) / sinA;
          }
          const tEnd = Math.max(Math.min(tMaxX, tMaxY) - 6, 105);

          return { ray, exitX, exitY, endX, endY, rayAngle, tEnd };
        });

        // 3. Vẽ 7 chùm tia sáng cầu vồng tỏa rộng xuyên suốt màn hình
        raysData.forEach(({ ray, exitX, exitY, endX, endY }) => {
          // Tia khúc xạ bên trong khối thủy tinh
          prismCtx.save();
          prismCtx.strokeStyle = ray.color;
          prismCtx.globalAlpha = 0.62;
          prismCtx.lineWidth = 2.2;
          prismCtx.beginPath();
          prismCtx.moveTo(hitX, hitY);
          prismCtx.lineTo(exitX, exitY);
          prismCtx.stroke();

          // Quầng sáng mềm rộng tạo hiệu ứng chùm tia quang phổ rực rỡ
          prismCtx.globalAlpha = 0.24;
          prismCtx.lineWidth = (isSmall ? 12 : 14) + prismPulse * 6;
          prismCtx.lineCap = 'round';
          prismCtx.beginPath();
          prismCtx.moveTo(exitX, exitY);
          prismCtx.lineTo(endX, endY);
          prismCtx.stroke();

          // Thân tia màu sắc nét
          prismCtx.globalAlpha = 0.84;
          prismCtx.lineWidth = (isSmall ? 4.2 : 5) + prismPulse * 2.5;
          prismCtx.shadowColor = ray.color;
          prismCtx.shadowBlur = 14 + prismPulse * 12;
          prismCtx.beginPath();
          prismCtx.moveTo(exitX, exitY);
          prismCtx.lineTo(endX, endY);
          prismCtx.stroke();

          // Lõi sáng trắng mảnh tạo cảm giác ánh sáng thật
          prismCtx.globalAlpha = 0.45;
          prismCtx.strokeStyle = '#ffffff';
          prismCtx.lineWidth = 1.4;
          prismCtx.shadowBlur = 0;
          prismCtx.beginPath();
          prismCtx.moveTo(exitX, exitY);
          prismCtx.lineTo(endX, endY);
          prismCtx.stroke();
          prismCtx.restore();
        });

        // 4. Vẽ chữ lời chúc và ngôi sao xoay trùng khít 100% dọc theo tâm của từng tia sáng
        raysData.forEach(({ ray, exitX, exitY, rayAngle, tEnd }) => {
          const fontSize = isSmall ? 13.5 : 15.5;

          prismCtx.save();
          prismCtx.translate(exitX, exitY);
          prismCtx.rotate(rayAngle);

          prismCtx.font = `700 ${fontSize}px Inter, sans-serif`;
          prismCtx.textAlign = 'right';
          prismCtx.textBaseline = 'middle';
          prismCtx.lineJoin = 'round';

          const textWidth = prismCtx.measureText(ray.name).width;
          const starX = tEnd - textWidth - (isSmall ? 11 : 13);

          // Viền tối mượt quanh chữ để chữ trắng nổi rõ ngay trên lõi tia sáng
          prismCtx.lineWidth = 5;
          prismCtx.strokeStyle = 'rgba(2, 6, 23, 0.94)';
          prismCtx.strokeText(ray.name, tEnd, 0);

          // Chữ trắng sáng bám đúng tâm tia (y = 0)
          prismCtx.fillStyle = '#ffffff';
          prismCtx.shadowColor = 'rgba(2, 6, 23, 0.9)';
          prismCtx.shadowBlur = 8;
          prismCtx.fillText(ray.name, tEnd, 0);

          // Ngôi sao màu quang phổ nằm chính tâm tia sáng ngay trước chữ
          prismCtx.textAlign = 'center';
          prismCtx.font = `700 ${fontSize + 1}px Inter, sans-serif`;
          prismCtx.lineWidth = 4.5;
          prismCtx.strokeStyle = 'rgba(2, 6, 23, 0.95)';
          prismCtx.shadowBlur = 0;
          prismCtx.strokeText('✦', starX, 0);
          prismCtx.fillStyle = ray.color;
          prismCtx.shadowColor = ray.color;
          prismCtx.shadowBlur = 10;
          prismCtx.fillText('✦', starX, 0);
          prismCtx.restore();
        });

        // 5. Vẽ khối Lăng Kính thủy tinh trong suốt chồng lên trên để tạo chiều sâu quang học
        prismCtx.save();
        const glassGrad = prismCtx.createLinearGradient(leftV.x, topV.y, rightV.x, rightV.y);
        glassGrad.addColorStop(0, 'rgba(224, 242, 254, 0.24)');
        glassGrad.addColorStop(0.5, 'rgba(56, 189, 248, 0.09)');
        glassGrad.addColorStop(1, 'rgba(251, 191, 36, 0.18)');

        prismCtx.beginPath();
        prismCtx.moveTo(topV.x, topV.y);
        prismCtx.lineTo(rightV.x, rightV.y);
        prismCtx.lineTo(leftV.x, leftV.y);
        prismCtx.closePath();
        prismCtx.fillStyle = glassGrad;
        prismCtx.fill();

        prismCtx.lineWidth = 2.2;
        prismCtx.strokeStyle = 'rgba(226, 232, 240, 0.7)';
        prismCtx.shadowColor = 'rgba(56, 189, 248, 0.45)';
        prismCtx.shadowBlur = 14;
        prismCtx.stroke();

        // Ghi chữ "Cô Thuý" tinh tế bên trong lăng kính
        prismCtx.shadowBlur = 0;
        prismCtx.font = `700 ${isSmall ? 13 : 15}px "Playfair Display", serif`;
        prismCtx.fillStyle = 'rgba(254, 243, 199, 0.92)';
        prismCtx.textAlign = 'center';
        prismCtx.fillText('Cô Thuý', cx, cy + triH * 0.22);
        prismCtx.restore();

        // Điểm sáng lóe lên nơi tia trắng chạm mặt lăng kính
        prismCtx.save();
        const flare = prismCtx.createRadialGradient(hitX, hitY, 1, hitX, hitY, 16 + prismPulse * 10);
        flare.addColorStop(0, 'rgba(255, 255, 255, 1)');
        flare.addColorStop(0.4, 'rgba(254, 240, 138, 0.65)');
        flare.addColorStop(1, 'rgba(255, 255, 255, 0)');
        prismCtx.fillStyle = flare;
        prismCtx.beginPath();
        prismCtx.arc(hitX, hitY, 16 + prismPulse * 10, 0, Math.PI * 2);
        prismCtx.fill();
        prismCtx.restore();
      }

      function setupPrismControls() {
        const incidentSlider = document.getElementById('incident-slider');
        const dispersionSlider = document.getElementById('dispersion-slider');
        const autoSweepBtn = document.getElementById('auto-sweep-btn');
        const sparkBtn = document.getElementById('prism-spark-btn');

        if (incidentSlider) {
          incidentSlider.addEventListener('input', (e) => {
            autoSweep = false;
            if (autoSweepBtn) autoSweepBtn.textContent = '🔄 Tự Động Quét';
            incidentAngleDeg = Number(e.target.value);
            updatePrismReadout();
          });
        }

        if (dispersionSlider) {
          dispersionSlider.addEventListener('input', (e) => {
            baseRefractiveIndex = Number(e.target.value);
            updatePrismReadout();
          });
        }

        if (autoSweepBtn) {
          autoSweepBtn.addEventListener('click', () => {
            autoSweep = !autoSweep;
            autoSweepBtn.textContent = autoSweep ? '⏸ Dừng Quét' : '🔄 Tự Động Quét';
            playChime('spark');
          });
        }

        if (sparkBtn) {
          sparkBtn.addEventListener('click', () => {
            prismPulse = 1.4;
            playChime('celebrate');
            const rect = prismContainer.getBoundingClientRect();
            launchFireworkBurst(rect.left + rect.width * 0.6, rect.top + rect.height * 0.45, 50);
            showToast('✨ Ánh sáng tri thức đang tỏa sáng rực rỡ tặng Cô Thuý!', 'success');
          });
        }

        // Hỗ trợ kéo chuột & cảm ứng trực tiếp trên canvas lăng kính
        let draggingPrism = false;
        function handlePrismPointer(clientY) {
          if (!prismContainer) return;
          const rect = prismContainer.getBoundingClientRect();
          const ratio = Math.min(Math.max((clientY - rect.top) / rect.height, 0), 1);
          autoSweep = false;
          if (autoSweepBtn) autoSweepBtn.textContent = '🔄 Tự Động Quét';
          incidentAngleDeg = 15 + ratio * 47;
          if (incidentSlider) incidentSlider.value = Math.round(incidentAngleDeg);
          updatePrismReadout();
        }

        if (prismContainer) {
          prismContainer.addEventListener('mousedown', (e) => {
            draggingPrism = true;
            handlePrismPointer(e.clientY);
          });

          window.addEventListener('mousemove', (e) => {
            pointer.x = e.clientX;
            pointer.y = e.clientY;
            if (draggingPrism) {
              handlePrismPointer(e.clientY);
            }
          });

          window.addEventListener('mouseup', () => {
            draggingPrism = false;
          });

          prismContainer.addEventListener(
            'touchstart',
            (e) => {
              if (e.touches && e.touches.length > 0) {
                e.preventDefault();
                draggingPrism = true;
                handlePrismPointer(e.touches[0].clientY);
              }
            },
            { passive: false }
          );

          prismContainer.addEventListener(
            'touchmove',
            (e) => {
              if (draggingPrism && e.touches && e.touches.length > 0) {
                e.preventDefault();
                handlePrismPointer(e.touches[0].clientY);
              }
            },
            { passive: false }
          );

          prismContainer.addEventListener('touchend', () => {
            draggingPrism = false;
          });
        }
      }

      function drawConservationSim(ctx, w, h, t) {
        // Nền lưới phòng thí nghiệm tinh tế
        ctx.strokeStyle = 'rgba(148, 163, 184, 0.06)';
        ctx.lineWidth = 1;
        for (let gx = 0; gx < w; gx += 26) {
          ctx.beginPath();
          ctx.moveTo(gx, 0);
          ctx.lineTo(gx, h);
          ctx.stroke();
        }
        for (let gy = 0; gy < h; gy += 26) {
          ctx.beginPath();
          ctx.moveTo(0, gy);
          ctx.lineTo(w, gy);
          ctx.stroke();
        }

        // Con lắc đơn dao động điều hòa chuyển hóa Wđ <-> Wt (hạ thấp điểm treo tránh bị nhãn góc trên che)
        const pivotX = w * 0.5;
        const pivotY = 44;
        const length = Math.min(h * 0.45, 108);
        const maxTheta = 0.66;
        const theta = maxTheta * Math.sin(t * 2.3);
        const bobX = pivotX + Math.sin(theta) * length;
        const bobY = pivotY + Math.cos(theta) * length;

        // Tỉ lệ Thế năng (Wt) và Động năng (Wđ) luôn có tổng = 1 (Bảo toàn)
        const wtRatio = Math.pow(Math.sin(t * 2.3), 2);
        const wdRatio = 1 - wtRatio;

        pendulumTrail.push({ x: bobX, y: bobY });
        if (pendulumTrail.length > 26) pendulumTrail.shift();

        // Quỹ đạo cung tròn mờ
        ctx.save();
        ctx.strokeStyle = 'rgba(251, 191, 36, 0.16)';
        ctx.setLineDash([4, 4]);
        ctx.lineWidth = 1.2;
        ctx.beginPath();
        ctx.arc(pivotX, pivotY, length, Math.PI * 0.5 - maxTheta, Math.PI * 0.5 + maxTheta);
        ctx.stroke();
        ctx.restore();

        // Vệt sáng chuyển động của quả cầu năng lượng
        for (let i = 1; i < pendulumTrail.length; i++) {
          const alpha = (i / pendulumTrail.length) * 0.55;
          ctx.beginPath();
          ctx.moveTo(pendulumTrail[i - 1].x, pendulumTrail[i - 1].y);
          ctx.lineTo(pendulumTrail[i].x, pendulumTrail[i].y);
          ctx.strokeStyle = `rgba(251, 191, 36, ${alpha})`;
          ctx.lineWidth = (i / pendulumTrail.length) * 7;
          ctx.lineCap = 'round';
          ctx.stroke();
        }

        // Dây treo & điểm treo
        ctx.strokeStyle = 'rgba(226, 232, 240, 0.65)';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(pivotX, pivotY);
        ctx.lineTo(bobX, bobY);
        ctx.stroke();

        ctx.fillStyle = '#fbbf24';
        ctx.beginPath();
        ctx.arc(pivotX, pivotY, 4.5, 0, Math.PI * 2);
        ctx.fill();

        // Quả cầu ánh sáng đổi sắc nhẹ giữa vàng (Wt) và xanh cyan (Wđ)
        ctx.save();
        const bobGrad = ctx.createRadialGradient(bobX - 3, bobY - 3, 2, bobX, bobY, 16);
        bobGrad.addColorStop(0, '#ffffff');
        bobGrad.addColorStop(0.45, wdRatio > 0.5 ? '#38bdf8' : '#fbbf24');
        bobGrad.addColorStop(1, wdRatio > 0.5 ? 'rgba(56, 189, 248, 0.25)' : 'rgba(245, 158, 11, 0.25)');
        ctx.fillStyle = bobGrad;
        ctx.shadowColor = wdRatio > 0.5 ? '#38bdf8' : '#fbbf24';
        ctx.shadowBlur = 18;
        ctx.beginPath();
        ctx.arc(bobX, bobY, 13, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();

        // Thanh đo Động năng (Wđ) + Thế năng (Wt) = Tổng năng lượng (100%) ở đáy khung
        const barY = h - 32;
        const barW = Math.min(w - 40, 320);
        const barX = (w - barW) / 2;
        const barH = 12;

        ctx.fillStyle = 'rgba(15, 23, 42, 0.9)';
        ctx.fillRect(barX, barY, barW, barH);

        const wdW = barW * wdRatio;
        ctx.fillStyle = '#38bdf8';
        ctx.fillRect(barX, barY, wdW, barH);
        ctx.fillStyle = '#fbbf24';
        ctx.fillRect(barX + wdW, barY, barW - wdW, barH);

        ctx.strokeStyle = 'rgba(255, 255, 255, 0.25)';
        ctx.lineWidth = 1;
        ctx.strokeRect(barX, barY, barW, barH);

        ctx.font = '600 11px Inter, sans-serif';
        ctx.textAlign = 'left';
        ctx.fillStyle = '#7dd3fc';
        ctx.fillText(`Wđ: ${Math.round(wdRatio * 100)}%`, barX, barY - 6);
        ctx.textAlign = 'center';
        ctx.fillStyle = '#e2e8f0';
        ctx.fillText('W = 100%', barX + barW / 2, barY - 6);
        ctx.textAlign = 'right';
        ctx.fillStyle = '#fde68a';
        ctx.fillText(`Wt: ${Math.round(wtRatio * 100)}%`, barX + barW, barY - 6);
      }

      function drawDispersionSim(ctx, w, h, t) {
        const prismX = w * 0.28;
        const prismY = h * 0.54;
        const size = Math.min(w * 0.2, h * 0.38);
        const triH = (Math.sqrt(3) / 2) * size;

        const topV = { x: prismX, y: prismY - triH * 0.6 };
        const leftV = { x: prismX - size / 2, y: prismY + triH * 0.4 };
        const rightV = { x: prismX + size / 2, y: prismY + triH * 0.4 };

        const hitX = topV.x + (leftV.x - topV.x) * 0.5;
        const hitY = topV.y + (leftV.y - topV.y) * 0.5;

        // Tia sáng trắng đập vào mặt trái
        ctx.save();
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.28)';
        ctx.lineWidth = 9;
        ctx.beginPath();
        ctx.moveTo(0, hitY + 18 + Math.sin(t * 1.4) * 8);
        ctx.lineTo(hitX, hitY);
        ctx.stroke();

        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 3;
        ctx.shadowColor = '#fef08a';
        ctx.shadowBlur = 12;
        ctx.beginPath();
        ctx.moveTo(0, hitY + 18 + Math.sin(t * 1.4) * 8);
        ctx.lineTo(hitX, hitY);
        ctx.stroke();
        ctx.restore();

        // 7 dải sóng quang phổ lan truyền kèm hạt photon lấp lánh
        const colors = ['#ff4d4d', '#ff944d', '#ffd633', '#4ade80', '#38bdf8', '#818cf8', '#d946ef'];
        colors.forEach((col, idx) => {
          const offset = idx - 3;
          const exitT = 0.48 + offset * 0.06;
          const ex = topV.x + (rightV.x - topV.x) * exitT;
          const ey = topV.y + (rightV.y - topV.y) * exitT;
          const targetY = h * 0.52 + offset * (h * 0.115);

          // Tia khúc xạ bên trong lăng kính
          ctx.beginPath();
          ctx.moveTo(hitX, hitY);
          ctx.lineTo(ex, ey);
          ctx.strokeStyle = col;
          ctx.globalAlpha = 0.65;
          ctx.lineWidth = 2;
          ctx.stroke();

          // Dải sóng màu sắc lan sang phải (bước sóng ngắn dần từ đỏ -> tím)
          ctx.save();
          ctx.globalAlpha = 0.9;
          ctx.strokeStyle = col;
          ctx.shadowColor = col;
          ctx.shadowBlur = 10;
          ctx.lineWidth = 2.8;
          ctx.beginPath();
          const steps = 42;
          const freq = 0.06 + idx * 0.015;
          for (let s = 0; s <= steps; s++) {
            const p = s / steps;
            const px = ex + (w - ex) * p;
            const baseY = ey + (targetY - ey) * p;
            const wave = Math.sin(px * freq - t * 5.2) * (3.2 * p);
            if (s === 0) ctx.moveTo(px, baseY + wave);
            else ctx.lineTo(px, baseY + wave);
          }
          ctx.stroke();

          // Hạt photon trượt dọc theo từng dải màu
          const photonP = (t * 0.48 + idx * 0.13) % 1;
          const phX = ex + (w - ex) * photonP;
          const phY = ey + (targetY - ey) * photonP + Math.sin(phX * freq - t * 5.2) * (3.2 * photonP);
          ctx.fillStyle = '#ffffff';
          ctx.beginPath();
          ctx.arc(phX, phY, 2.8, 0, Math.PI * 2);
          ctx.fill();
          ctx.restore();
        });
        ctx.globalAlpha = 1;

        // Vẽ khối lăng kính thủy tinh
        ctx.save();
        const glassGrad = ctx.createLinearGradient(leftV.x, topV.y, rightV.x, rightV.y);
        glassGrad.addColorStop(0, 'rgba(224, 242, 254, 0.28)');
        glassGrad.addColorStop(1, 'rgba(56, 189, 248, 0.12)');
        ctx.beginPath();
        ctx.moveTo(topV.x, topV.y);
        ctx.lineTo(rightV.x, rightV.y);
        ctx.lineTo(leftV.x, leftV.y);
        ctx.closePath();
        ctx.fillStyle = glassGrad;
        ctx.fill();
        ctx.strokeStyle = 'rgba(226, 232, 240, 0.8)';
        ctx.lineWidth = 2;
        ctx.stroke();
        ctx.restore();
      }

      function drawGravitySim(ctx, w, h, t) {
        const cx = w * 0.5;
        const cy = h * 0.52;

        // Lưới không-thời gian uốn cong nhẹ về tâm hấp dẫn
        ctx.save();
        ctx.strokeStyle = 'rgba(139, 92, 246, 0.18)';
        ctx.lineWidth = 1;
        for (let r = 28; r <= Math.max(w, h) * 0.55; r += 28) {
          ctx.beginPath();
          ctx.ellipse(cx, cy, r * 1.45, r * 0.62, 0, 0, Math.PI * 2);
          ctx.stroke();
        }
        ctx.restore();

        // 4 hành tinh học trò quay quanh mặt trời tri thức Cô Thuý
        const orbits = [
          { rx: w * 0.18, ry: h * 0.2, speed: 1.55, phase: 0.4, color: '#38bdf8', r: 5 },
          { rx: w * 0.27, ry: h * 0.29, speed: 1.12, phase: 2.1, color: '#c084fc', r: 6 },
          { rx: w * 0.36, ry: h * 0.36, speed: 0.82, phase: 4.0, color: '#4ade80', r: 5.5 },
          { rx: w * 0.43, ry: h * 0.42, speed: 0.62, phase: 5.4, color: '#fb7185', r: 4.8 }
        ];

        orbits.forEach((orb) => {
          const angle = t * orb.speed + orb.phase;
          const px = cx + Math.cos(angle) * orb.rx;
          const py = cy + Math.sin(angle) * orb.ry;

          // Đường lực hấp dẫn hướng về tâm
          ctx.save();
          ctx.strokeStyle = 'rgba(251, 191, 36, 0.22)';
          ctx.setLineDash([3, 4]);
          ctx.lineWidth = 1;
          ctx.beginPath();
          ctx.moveTo(px, py);
          ctx.lineTo(cx, cy);
          ctx.stroke();
          ctx.restore();

          // Đuôi sao sáng trên quỹ đạo
          for (let k = 1; k <= 10; k++) {
            const prevAngle = angle - k * 0.08;
            const tx = cx + Math.cos(prevAngle) * orb.rx;
            const ty = cy + Math.sin(prevAngle) * orb.ry;
            ctx.beginPath();
            ctx.arc(tx, ty, orb.r * (1 - k / 12) * 0.65, 0, Math.PI * 2);
            ctx.fillStyle = orb.color;
            ctx.globalAlpha = (1 - k / 11) * 0.4;
            ctx.fill();
          }
          ctx.globalAlpha = 1;

          // Quả cầu hành tinh
          ctx.save();
          ctx.fillStyle = orb.color;
          ctx.shadowColor = orb.color;
          ctx.shadowBlur = 12;
          ctx.beginPath();
          ctx.arc(px, py, orb.r, 0, Math.PI * 2);
          ctx.fill();
          ctx.restore();
        });

        // Mặt trời trung tâm "Cô Thuý" phát hào quang ấm áp
        ctx.save();
        const pulse = 1 + Math.sin(t * 2.5) * 0.06;
        const sunGrad = ctx.createRadialGradient(cx, cy, 4, cx, cy, 34 * pulse);
        sunGrad.addColorStop(0, '#fef9c3');
        sunGrad.addColorStop(0.45, '#f59e0b');
        sunGrad.addColorStop(1, 'rgba(245, 158, 11, 0)');
        ctx.fillStyle = sunGrad;
        ctx.beginPath();
        ctx.arc(cx, cy, 34 * pulse, 0, Math.PI * 2);
        ctx.fill();

        ctx.fillStyle = '#fffbeb';
        ctx.font = '700 11px "Playfair Display", serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.shadowColor = 'rgba(2, 6, 23, 0.9)';
        ctx.shadowBlur = 6;
        ctx.fillText('Cô Thuý', cx, cy);
        ctx.restore();
      }

      function drawResonanceSim(ctx, w, h, t) {
        const midY = h * 0.54;
        // Mức độ đồng bộ tần số dao động mượt từ 0 -> 1 -> 0 tạo hiệu ứng cộng hưởng rõ rệt
        const sync = 0.5 + 0.5 * Math.sin(t * 1.15 - Math.PI / 2);
        const resonanceBoost = Math.pow(sync, 2.2);

        // Đường trục cân bằng
        ctx.strokeStyle = 'rgba(148, 163, 184, 0.18)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(0, midY);
        ctx.lineTo(w, midY);
        ctx.stroke();

        // Sóng 1: Tần số truyền lửa của Cô (f₀ - màu vàng)
        ctx.save();
        ctx.strokeStyle = 'rgba(251, 191, 36, 0.45)';
        ctx.lineWidth = 1.8;
        ctx.beginPath();
        for (let x = 0; x <= w; x += 4) {
          const y = midY + Math.sin(x * 0.035 - t * 3.8) * 20;
          if (x === 0) ctx.moveTo(x, y);
          else ctx.lineTo(x, y);
        }
        ctx.stroke();

        // Sóng 2: Tần số học trò (f - màu xanh cyan) tiến dần về f₀
        const studentFreq = 0.035 + (1 - sync) * 0.022;
        ctx.strokeStyle = 'rgba(56, 189, 248, 0.45)';
        ctx.lineWidth = 1.8;
        ctx.beginPath();
        for (let x = 0; x <= w; x += 4) {
          const y = midY + Math.sin(x * studentFreq - t * 3.8) * 20;
          if (x === 0) ctx.moveTo(x, y);
          else ctx.lineTo(x, y);
        }
        ctx.stroke();

        // Sóng tổng hợp cộng hưởng bùng nổ biên độ A -> A_max
        const maxAmp = 24 + resonanceBoost * (h * 0.25);
        const resGrad = ctx.createLinearGradient(0, 0, w, 0);
        resGrad.addColorStop(0, '#fb7185');
        resGrad.addColorStop(0.5, '#fbbf24');
        resGrad.addColorStop(1, '#e879f9');

        ctx.strokeStyle = resGrad;
        ctx.lineWidth = 3 + resonanceBoost * 2.2;
        ctx.shadowColor = '#fb7185';
        ctx.shadowBlur = 10 + resonanceBoost * 16;
        ctx.beginPath();
        for (let x = 0; x <= w; x += 3) {
          const env = Math.sin((x / w) * Math.PI);
          const wave = Math.sin(x * 0.035 - t * 3.8);
          const y = midY + wave * maxAmp * (0.45 + 0.55 * env);
          if (x === 0) ctx.moveTo(x, y);
          else ctx.lineTo(x, y);
        }
        ctx.stroke();
        ctx.restore();

        // Nhãn trạng thái tần số trực quan ở đáy khung
        ctx.font = '600 11.5px Inter, sans-serif';
        ctx.textAlign = 'center';
        ctx.fillStyle = resonanceBoost > 0.72 ? '#fda4af' : '#cbd5e1';
        const statusText =
          resonanceBoost > 0.72
            ? '✨ Cộng hưởng cực đại! (f = f₀ ⇒ A → A_max)'
            : `Đang đồng bộ tần số Cô & Trò: ${Math.round(sync * 100)}%`;
        ctx.fillText(statusText, w * 0.5, h - 14);
      }

      function renderLawModalFrame() {
        if (!activeLawKey || !lawModalCtx) return;
        lawSimTime += 0.016;

        const w = lawModalWidth;
        const h = lawModalHeight;
        lawModalCtx.clearRect(0, 0, w, h);

        if (activeLawKey === 'conservation') {
          drawConservationSim(lawModalCtx, w, h, lawSimTime);
        } else if (activeLawKey === 'dispersion') {
          drawDispersionSim(lawModalCtx, w, h, lawSimTime);
        } else if (activeLawKey === 'gravity') {
          drawGravitySim(lawModalCtx, w, h, lawSimTime);
        } else if (activeLawKey === 'resonance') {
          drawResonanceSim(lawModalCtx, w, h, lawSimTime);
        }
      }

      function openLawModal(lawKey) {
        const data = LAW_EXPERIENCES[lawKey];
        if (!data || !lawModalEl) return;

        activeLawKey = lawKey;
        lawSimTime = 0;
        pendulumTrail = [];

        const iconEl = document.getElementById('law-modal-icon');
        const formulaEl = document.getElementById('law-modal-formula');
        const borderEl = document.getElementById('law-modal-border');
        const titleEl = document.getElementById('law-modal-title');
        const simBadgeEl = document.getElementById('law-modal-sim-badge');
        const wishEl = document.getElementById('law-modal-wish');
        const extraEl = document.getElementById('law-modal-extra');

        if (iconEl) iconEl.textContent = data.icon;
        if (formulaEl) {
          formulaEl.textContent = data.formula;
          formulaEl.className = `inline-block px-2.5 py-0.5 rounded-lg bg-slate-950/90 border border-white/10 font-mono text-xs mb-1 ${data.formulaClass}`;
        }
        if (borderEl) {
          borderEl.className = `relative w-full max-w-xl max-h-[90vh] flex flex-col rounded-3xl p-[1.5px] bg-gradient-to-br ${data.borderClass} shadow-2xl`;
        }
        if (titleEl) titleEl.textContent = data.title;
        if (simBadgeEl) simBadgeEl.textContent = data.simBadge;
        if (wishEl) wishEl.textContent = data.wish;
        if (extraEl) extraEl.textContent = data.extra;

        // Đánh dấu thẻ định luật này đã được cô xem
        const viewedBadge = document.querySelector(`[data-law-viewed="${lawKey}"]`);
        if (viewedBadge) viewedBadge.classList.remove('hidden');

        lawModalEl.classList.remove('hidden');
        lawModalEl.classList.add('flex');
        document.body.style.overflow = 'hidden';

        // Đồng bộ kích thước canvas ngay lập tức và xác nhận lại ở khung hình kế tiếp
        resizeLawModalCanvas();
        requestAnimationFrame(() => {
          resizeLawModalCanvas();
        });

        playChime('spark');
        launchFireworkBurst(window.innerWidth * 0.5, window.innerHeight * 0.25, 30);
      }

      function closeLawModal() {
        if (!lawModalEl) return;
        activeLawKey = null;
        lawModalEl.classList.add('hidden');
        lawModalEl.classList.remove('flex');
        document.body.style.overflow = '';
      }

      function setupLawModal() {
        const openButtons = document.querySelectorAll('[data-open-law]');
        openButtons.forEach((btn) => {
          btn.addEventListener('click', () => {
            const key = btn.getAttribute('data-open-law');
            if (key) openLawModal(key);
          });
        });

        const closeBtn = document.getElementById('law-modal-close-btn');
        if (closeBtn) {
          closeBtn.addEventListener('click', closeLawModal);
        }

        const sparkBtn = document.getElementById('law-modal-spark-btn');
        if (sparkBtn) {
          sparkBtn.addEventListener('click', () => {
            triggerCelebrationSalvo();
          });
        }

        // Chạm ra vùng tối bên ngoài hộp thoại cũng đóng gọn
        if (lawModalEl) {
          lawModalEl.addEventListener('click', (e) => {
            if (e.target === lawModalEl) closeLawModal();
          });
        }

        // Hỗ trợ phím Escape trên máy tính
        window.addEventListener('keydown', (e) => {
          if (e.key === 'Escape' && activeLawKey) {
            closeLawModal();
          }
        });
      }

      // Khởi chạy toàn bộ ứng dụng sau khi trang tải hoàn chỉnh
      let appInitialized = false;
      function startApp() {
        if (appInitialized) return;
        appInitialized = true;

        try {
          resizeBgCanvas();
          resizeFwCanvas();
          resizePrismCanvas();
          updatePrismReadout();

          setupSoundToggle();
          setupGreetingCard();
          setupPrismControls();
          setupQuantumGifts();
          setupLawModal();
          setupExtras();

          // Tự động cập nhật kích thước các canvas khi thay đổi màn hình / xoay điện thoại
          let resizeTimer = null;
          window.addEventListener('resize', () => {
            clearTimeout(resizeTimer);
            resizeTimer = setTimeout(() => {
              resizeBgCanvas();
              resizeFwCanvas();
              resizePrismCanvas();
              resizeLawModalCanvas();
            }, 120);
          });

          // Bắt đầu vòng lặp đồ họa
          animateLoop();

          // Một chùm pháo hoa chào mừng nhẹ nhàng sau khi vào trang
          setTimeout(() => {
            launchFireworkBurst(window.innerWidth * 0.5, window.innerHeight * 0.25, 36);
          }, 700);
        } catch (err) {
          console.error('Lỗi khởi tạo trang web 20/11:', err);
        }
      }

      window.onload = function () {
        startApp();
      };

      // Dự phòng trường hợp DOM đã sẵn sàng trước khi sự kiện onload kích hoạt trong khung xem trước
      if (document.readyState === 'interactive' || document.readyState === 'complete') {
        startApp();
      }
    })();
